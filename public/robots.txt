# robots.txt — thegrowthpmm.com
#
# Intent: let search engines and AI *answer* engines read and cite the site
# (they send traffic and surface the brand), while opting out of AI *training*
# scrapers (which copy content but send nothing back).
#
# NOTE: This file only takes effect if Cloudflare's "Manage your robots.txt"
# setting is turned OFF. While that setting is on, Cloudflare generates and
# serves its own robots.txt and this file is ignored.

# ---------------------------------------------------------------------------
# Default: every bot may crawl everything. Regular search crawlers
# (Googlebot, Bingbot, DuckDuckBot, etc.) are covered here and need no
# entry of their own.
# ---------------------------------------------------------------------------
User-agent: *
Allow: /

# ---------------------------------------------------------------------------
# AI answer / search engines — explicitly welcome. These read the site to
# answer users live and cite/link back, so they aid discovery.
# ---------------------------------------------------------------------------

# OpenAI — ChatGPT search results
User-agent: OAI-SearchBot
Allow: /

# OpenAI — ChatGPT live browsing (triggered when a user asks it to fetch a page)
User-agent: ChatGPT-User
Allow: /

# Perplexity — answer index
User-agent: PerplexityBot
Allow: /

# Perplexity — live user-triggered fetches
User-agent: Perplexity-User
Allow: /

# Anthropic — Claude web search / citations
User-agent: ClaudeBot
Allow: /

# Apple — Siri / Spotlight search (NOT the -Extended training variant below)
User-agent: Applebot
Allow: /

# ---------------------------------------------------------------------------
# AI *training* scrapers — opted out. These harvest content to train models
# and generally don't cite or drive traffic back.
# ---------------------------------------------------------------------------

# OpenAI — model training crawler

Disallow: /

# Google — Gemini model training (separate from Googlebot search, which stays allowed)

# Common Crawl — feeds many third-party training datasets

# ByteDance / TikTok scraper

Disallow: /

# Meta — AI training crawler

Disallow: /

# Amazon — broad AI/Alexa crawler
